## 7. User Flow

1. **Customer Onboarding & Portal Access:** Customer signs up or logs in. They can view their orders, active support tickets, and AI-curated product recommendations based on past interactions.
2. **AI-Assisted Resolution Flow:** Customer opens the `/portal/chat` widget or creates a support ticket.
3. **Automated AI Triage Engine:** Node.js backend passes customer input to the `@google/genai` engine. Gemini classifies intent, assesses sentiment, assigns priority score, checks knowledge base, and formulates an accurate response.
4. **Deterministic Escalation:** If sentiment score falls below critical thresholds or intent indicates churn/frustration, Gemini returns an `ESCALATE` action.
5. **Agent Copilot Handoff:** The ticket instantly drops into `/admin/tickets`. The agent opens the detail page and views an AI-synthesized handover summary, customer sentiment trend, and 3 pre-generated contextual responses.
6. **Resolution & Feedback Loop:** Once marked resolved, the system prompts the customer for feedback via `/portal/feedback`. Sentiment and insights are processed into the executive `/admin/insights` dashboard.

## 10. Database Schema (Production SQL for PostgreSQL)

Execute the following DDL script to create the relational schema:

```sql
-- Enable UUID Extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. USERS TABLE
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    role VARCHAR(20) NOT NULL DEFAULT 'customer' CHECK (role IN ('customer', 'agent', 'admin')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. CUSTOMER PROFILES (360 Degree View)
CREATE TABLE customer_profiles (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID UNIQUE REFERENCES users(id) ON DELETE CASCADE,
    company_name VARCHAR(150),
    lifetime_value DECIMAL(10, 2) DEFAULT 0.00,
    churn_risk_score DECIMAL(3, 2) DEFAULT 0.00, -- Range 0.00 to 1.00
    overall_sentiment_score DECIMAL(3, 2) DEFAULT 0.50, -- Range 0.00 to 1.00
    preferred_channel VARCHAR(30) DEFAULT 'web_chat',
    metadata JSONB DEFAULT '{}'::jsonb,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. TICKETS TABLE
CREATE TABLE tickets (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    ticket_number SERIAL UNIQUE,
    customer_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    assigned_agent_id UUID REFERENCES users(id) ON DELETE SET NULL,
    subject VARCHAR(255) NOT NULL,
    description TEXT NOT NULL,
    status VARCHAR(30) DEFAULT 'open' CHECK (status IN ('open', 'ai_handling', 'in_progress', 'escalated', 'resolved', 'closed')),
    priority VARCHAR(20) DEFAULT 'medium' CHECK (priority IN ('low', 'medium', 'high', 'critical')),
    domain_category VARCHAR(50) NOT NULL,
    sentiment_label VARCHAR(20) DEFAULT 'neutral' CHECK (sentiment_label IN ('positive', 'neutral', 'negative', 'frustrated')),
    ai_summary TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 4. TICKET MESSAGES TABLE
CREATE TABLE ticket_messages (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    ticket_id UUID NOT NULL REFERENCES tickets(id) ON DELETE CASCADE,
    sender_id UUID REFERENCES users(id) ON DELETE SET NULL,
    sender_type VARCHAR(20) NOT NULL CHECK (sender_type IN ('customer', 'agent', 'ai_assistant', 'system')),
    message TEXT NOT NULL,
    sentiment_score DECIMAL(3, 2),
    is_internal_note BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 5. FEEDBACK TABLE
CREATE TABLE customer_feedback (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    ticket_id UUID REFERENCES tickets(id) ON DELETE SET NULL,
    customer_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    csat_rating INT NOT NULL CHECK (csat_rating BETWEEN 1 AND 5),
    nps_score INT CHECK (nps_score BETWEEN 0 AND 10),
    qualitative_feedback TEXT,
    extracted_topics JSONB DEFAULT '[]'::jsonb,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- INDEXES FOR HIGH-PERFORMANCE QUERYING
CREATE INDEX idx_tickets_customer ON tickets(customer_id);
CREATE INDEX idx_tickets_status ON tickets(status);
CREATE INDEX idx_tickets_priority ON tickets(priority);
CREATE INDEX idx_messages_ticket ON ticket_messages(ticket_id);
CREATE INDEX idx_feedback_customer ON customer_feedback(customer_id);

```

## 11. Row Level Security (RLS) / Data Isolation Rules

* **Customers:**
* Can strictly read and update ONLY their own user record, customer profile, submitted support tickets, associated ticket messages, and feedback entries.
* Attempting to query `/api/tickets/:id` where `ticket.customer_id != req.user.id` must return an immediate HTTP `403 Forbidden`.


* **Support Agents & Admins:**
* Can read and modify ALL tickets, customer profiles, system analytics, and feedback records.
* Internal notes (`is_internal_note = true`) are stripped out completely before returning message payloads to non-agent/customer callers.

## 16. Zod Validation Requirements

// backend/validators/cxValidators.js
import { z } from 'zod';

export const registerSchema = z.object({
  email: z.string().email('Invalid email address format'),
  password: z.string().min(8, 'Password must be at least 8 characters long'),
  full_name: z.string().min(2, 'Full name is required'),
  role: z.enum(['customer', 'agent', 'admin']).default('customer')
});

export const createTicketSchema = z.object({
  subject: z.string().min(5, 'Subject must be at least 5 characters').max(255),
  description: z.string().min(15, 'Description must provide sufficient detail (min 15 chars)'),
  domain_category: z.enum(['ecommerce', 'saas', 'fintech', 'healthcare', 'general']),
  priority: z.enum(['low', 'medium', 'high', 'critical']).optional()
});

export const messageSchema = z.object({
  message: z.string().min(1, 'Message cannot be empty'),
  is_internal_note: z.boolean().optional().default(false)
});

export const feedbackSchema = z.object({
  ticket_id: z.string().uuid().optional(),
  csat_rating: z.number().int().min(1).max(5),
  nps_score: z.number().int().min(0).max(10).optional(),
  qualitative_feedback: z.string().optional()
});
